-- Create the database
CREATE DATABASE IF NOT EXISTS Building_Maintenance;

-- Use the created database
USE Building_Maintenance;

-- Create the Tenant_details table
CREATE TABLE IF NOT EXISTS Tenant_details (
    Tenant_id INT PRIMARY KEY AUTO_INCREMENT, -- Primary key for unique tenant identification
    Tenant_name VARCHAR(100) NOT NULL,
    Room_number INT NOT NULL UNIQUE,
    Mobile_number VARCHAR(15) NOT NULL,
    Permanent_address TEXT NOT NULL,
    Local_address TEXT NOT NULL,
    Aadhaar_number VARCHAR(12) NOT NULL UNIQUE
);
